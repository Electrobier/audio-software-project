cmake_minimum_required(VERSION 3.22)
project(IRON2500 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(IRON_DSP_ONLY "Build only portable DSP tests" OFF)
include(CTest)
add_executable(iron_dsp_tests Tests/DspTests.cpp)
target_include_directories(iron_dsp_tests PRIVATE Source)
add_test(NAME DSP COMMAND iron_dsp_tests)
if(IRON_DSP_ONLY)
  return()
endif()
if(NOT DEFINED JUCE_SOURCE_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/JUCE/CMakeLists.txt")
  set(JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/JUCE")
endif()
if(DEFINED JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(IRON2500
  COMPANY_NAME "Iron Instruments" BUNDLE_ID "audio.ironinstruments.iron2500"
  PLUGIN_MANUFACTURER_CODE Irni PLUGIN_CODE Ir25
  FORMATS VST3 Standalone PRODUCT_NAME "IRON 2500"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  VST3_CATEGORIES Fx Dynamics COPY_PLUGIN_AFTER_BUILD FALSE)
target_include_directories(IRON2500 PUBLIC Source "${JUCE_SOURCE_DIR}/modules")
target_sources(IRON2500 PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(IRON2500 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(IRON2500 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(iron_state_tests Tests/StateTests.cpp)
  target_link_libraries(iron_state_tests PRIVATE IRON2500)
  add_test(NAME State COMMAND iron_state_tests)
endif()
