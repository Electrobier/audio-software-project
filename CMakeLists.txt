cmake_minimum_required(VERSION 3.22)
project(White224 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WHITE224_BUILD_PLUGIN "Build JUCE VST3 and Standalone" ON)
option(WHITE224_SANITIZE "Enable address/undefined sanitizers on core tests" OFF)
add_library(white224_dsp src/dsp/ReverbEngine.cpp src/presets/FactoryPresets.cpp)
target_include_directories(white224_dsp PUBLIC src)
set_target_properties(white224_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(white224_dsp PRIVATE /W4 /fp:precise)
else()
 target_compile_options(white224_dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
add_executable(white224_tests tests/DspTests.cpp)
target_link_libraries(white224_tests PRIVATE white224_dsp)
if(WHITE224_SANITIZE AND NOT MSVC)
 target_compile_options(white224_dsp PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(white224_dsp PUBLIC -fsanitize=address,undefined)
endif()
add_test(NAME DSP COMMAND white224_tests)
if(WHITE224_BUILD_PLUGIN)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE/CMakeLists.txt" AND NOT DEFINED JUCE_SOURCE_DIR)
  set(JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE")
 endif()
 if(DEFINED JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
 else()
  include(FetchContent)
  FetchContent_Declare(JUCE URL https://codeload.github.com/juce-framework/JUCE/tar.gz/refs/tags/8.0.6 URL_HASH SHA256=95752f8e8b37d802b4c10182bc757de7d88f0c8899e879afa8798589be306a11)
  FetchContent_MakeAvailable(JUCE)
  set(JUCE_SOURCE_DIR "${juce_SOURCE_DIR}")
 endif()
 juce_add_plugin(White224 COMPANY_NAME "White224 Audio" BUNDLE_ID "audio.white224.reverb"
  PLUGIN_MANUFACTURER_CODE WhtA PLUGIN_CODE W224 FORMATS VST3 Standalone
  PRODUCT_NAME "White224 Vintage Reverb" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Reverb)
 juce_generate_juce_header(White224)
 target_sources(White224 PRIVATE src/PluginProcessor.cpp src/PluginEditor.cpp)
 target_compile_definitions(White224 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(White224 PRIVATE white224_dsp juce::juce_audio_utils juce::juce_audio_plugin_client PUBLIC juce::juce_recommended_config_flags)
 add_executable(white224_state_tests tests/StateTests.cpp)
 target_link_libraries(white224_state_tests PRIVATE White224)
 target_include_directories(white224_state_tests PRIVATE src "${JUCE_SOURCE_DIR}/modules" "${CMAKE_BINARY_DIR}/White224_artefacts/JuceLibraryCode")
 target_compile_definitions(white224_state_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 add_test(NAME State COMMAND white224_state_tests)
endif()
