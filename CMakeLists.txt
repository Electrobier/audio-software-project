cmake_minimum_required(VERSION 3.22)
project(Psy303 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/utf-8)
endif()
option(PSY303_PLUGIN "Build JUCE VST3 and standalone" ON)
add_library(psycore INTERFACE)
target_include_directories(psycore INTERFACE Source)
add_executable(psy303_tests Tests/Tests.cpp)
target_link_libraries(psy303_tests PRIVATE psycore)
add_executable(psy303_render tools/Render.cpp)
target_link_libraries(psy303_render PRIVATE psycore)
enable_testing()
add_test(NAME core COMMAND psy303_tests)
if(PSY303_PLUGIN)
 if(DEFINED JUCE_DIR)
  find_package(JUCE CONFIG REQUIRED)
 elseif(DEFINED JUCE_PATH)
  add_subdirectory("${JUCE_PATH}" JUCE)
 elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/JUCE/CMakeLists.txt")
  add_subdirectory(ThirdParty/JUCE JUCE)
 else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
 endif()
 if(CMAKE_CROSSCOMPILING)
  set(manifest FALSE)
 else()
  set(manifest TRUE)
 endif()
 juce_add_plugin(Psy303 VST3_AUTO_MANIFEST ${manifest} COMPANY_NAME "Psy303 Audio" BUNDLE_ID "audio.psy303.bassline" PLUGIN_MANUFACTURER_CODE PsyA PLUGIN_CODE Ps03 FORMATS VST3 Standalone PRODUCT_NAME "Psy303 Bassline" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
 target_sources(Psy303 PRIVATE Source/Plugin.cpp Source/Editor.cpp)
 target_link_libraries(Psy303 PRIVATE psycore juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
 target_compile_definitions(Psy303 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endif()
add_executable(psy303_diagnostics tools/Diagnostics.cpp)
target_link_libraries(psy303_diagnostics PRIVATE psycore)
if(PSY303_PLUGIN)
 add_executable(psy303_plugin_tests Tests/PluginTests.cpp)
 target_include_directories(psy303_plugin_tests PRIVATE $<TARGET_PROPERTY:Psy303,INCLUDE_DIRECTORIES>)
 target_compile_definitions(psy303_plugin_tests PRIVATE $<TARGET_PROPERTY:Psy303,COMPILE_DEFINITIONS>)
 target_link_libraries(psy303_plugin_tests PRIVATE Psy303)
 add_test(NAME plugin COMMAND psy303_plugin_tests)
endif()
